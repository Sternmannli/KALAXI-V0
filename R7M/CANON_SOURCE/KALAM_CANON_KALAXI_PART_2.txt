    Sign data with Ed25519. If private_key_pem is None, return a mock signature.
    """
    if not CRYPTO_AVAILABLE or private_key_pem is None:
        # mock signature (for testing only)
        return "mock_signature_" + hashlib.sha256(data).hexdigest()[:16]

    private_key = serialization.load_pem_private_key(
        private_key_pem,
        password=None
    )
    signature = private_key.sign(data)
    return signature.hex()

def verify_signature(data: bytes, signature: str, public_key_pem: bytes) -> bool:
    """
    Verify Ed25519 signature. If crypto not available, mock verification (always True).
    """
    if not CRYPTO_AVAILABLE:
        return True  # mock
    public_key = serialization.load_pem_public_key(public_key_pem)
    try:
        public_key.verify(bytes.fromhex(signature), data)
        return True
    except Exception:
        return False

def canonical_json(obj: Any) -> bytes:
    """Return canonical JSON bytes (RFC‑8785)."""
    if JCS_AVAILABLE:
        return jcs.canonicalize(obj)
    else:
        # fallback: standard JSON with sorted keys (not fully canonical)
        return json.dumps(obj, sort_keys=True, separators=(',', ':')).encode('utf-8')

        self._tokens = {}           # token -> (institution_id, expiry)
        self._bundles = []           # list of stored bundles (for mirror aggregation)

    def handle_offer(self, offer: dict) -> Tuple[int, Optional[dict]]:
        """
        Process an offer. Returns (status_code, response_body).
        """
        # 1. Basic validation
        required = ["institution_id", "period_from", "period_to", "contribution_score", "bundle_hash", "signature"]
        if not all(k in offer for k in required):
            return 400, {"error": "Missing fields"}

        # 2. Verify contribution score
        if offer["contribution_score"] < self.policy["contribution_min"]:
            return 403, {"error": "Insufficient contribution score"}

        # 3. Verify signature (stub)
        data = canonical_json({k: offer[k] for k in required if k != "signature"})
        pub_key = directory.get_public_key(offer["institution_id"])
        if pub_key is None:
            return 403, {"error": "Unknown institution"}
        if not verify_signature(data, offer["signature"], pub_key):
            return 403, {"error": "Invalid signature"}

        # 4. Accept – generate token
        token = secrets.token_urlsafe(32)
        expiry = time.time() + TOKEN_EXPIRY_SECONDS
        self._tokens[token] = (offer["institution_id"], expiry)

        return 202, {
            "exchange_token": token,
            "policy": self.policy
        }

    def handle_bundle_submission(self, submission: dict) -> Tuple[int, Optional[dict]]:
        """
        Process a bundle submission. Returns (status_code, response_body).
        """
        # 1. Validate token
        token = submission.get("exchange_token")
        if token not in self._tokens:
            return 403, {"error": "Invalid or expired token"}
        inst_id, expiry = self._tokens[token]
        if time.time() > expiry:
            del self._tokens[token]
            return 403, {"error": "Token expired"}

        # 2. Check bundle fields
        bundle = submission.get("bundle")
        if not bundle:
            return 400, {"error": "Missing bundle"}

        # 3. Verify hash
        computed_hash = hashlib.sha256(canonical_json(bundle)).hexdigest()
        if computed_hash != submission.get("hash"):
            return 400, {"error": "Hash mismatch"}

        # 4. Verify signature (stub)
        pub_key = directory.get_public_key(inst_id)
        if pub_key and not verify_signature(canonical_json(bundle), submission.get("signature", ""), pub_key):
            return 403, {"error": "Invalid signature"}

        # 5. Store bundle (append to list)
        self._bundles.append(bundle)

        # 6. Consume token (one‑time use)
        del self._tokens[token]

        return 201, {"status": "accepted", "receipt_id": hashlib.sha256(canonical_json(bundle)).hexdigest()[:16]}

    def generate_mirror(self, institution_id: str, period_from: str, period_to: str) -> Optional[CounterMirror]:

            period_from=period_from,
            period_to=period_to,

        mirror.signature = sign_data(data, private_key_pem=None)  # mock

# ----------------------------------------------------------------------
# Laplace noise for differential privacy
# ----------------------------------------------------------------------
def laplace_noise(scale: float) -> float:
    """Generate Laplace noise with given scale."""
    import random, math
    u = random.random() - 0.5
    return -scale * math.copysign(math.log(1 - 2 * abs(u)), u)

def add_noise(count: int, epsilon: float, sensitivity: float = 1.0) -> int:
    """Add Laplace noise to a count."""
    scale = sensitivity / epsilon
    noisy = count + laplace_noise(scale)
    return max(0, int(round(noisy)))

# ----------------------------------------------------------------------
# Core aggregation
# ----------------------------------------------------------------------
def aggregate_bundles(
    bundles: List[dict],
    exclude_institution: Optional[str] = None,
    k_min: int = DEFAULT_K_MIN,
    epsilon: float = DEFAULT_EPSILON,
    top_n: int = 25
) -> List[dict]:
    """
    Aggregate patterns from a list of essence bundles.
    - bundles: list of bundle dictionaries (as stored after submission).
    - exclude_institution: if provided, skip bundles from this institution (the requester).
    - k_min, epsilon: privacy parameters.
    - top_n: maximum number of patterns to return.
    Returns a list of aggregated pattern items, each with:
        pattern: str
        national_count: int (noisy, k‑anonymized)
        median_mttr_days: float (optional, not yet implemented)
    """
    # Step 1: collect all patterns
    pattern_counts = defaultdict(int)
    pattern_mttrs = defaultdict(list)

    for bundle in bundles:
        inst_id = bundle.get("institution_id")
        if exclude_institution and inst_id == exclude_institution:
            continue
        items = bundle.get("items", [])
        for item in items:
            pat = item.get("pattern")
            cnt = item.get("count", 0)
            mttr = item.get("mttr_days")
            if pat:
                pattern_counts[pat] += cnt
                if mttr is not None:
                    pattern_mttrs[pat].append(mttr)

    # Step 2: filter by k‑anonymity and apply DP noise
    aggregated = []
    for pat, total_count in pattern_counts.items():
        if total_count >= k_min:
            noisy = add_noise(total_count, epsilon)
            # compute median MTTR if available
            mttr_list = pattern_mttrs.get(pat, [])
            if mttr_list:
                mttr_list.sort()
                n = len(mttr_list)
                median = mttr_list[n // 2] if n % 2 else (mttr_list[n//2 - 1] + mttr_list[n//2]) / 2
            else:
                median = None
            aggregated.append({
                "pattern": pat,
                "national_count": noisy,
                "median_mttr_days": round(median, 1) if median is not None else None
            })

    # Step 3: sort by count (descending) and take top_n
    aggregated.sort(key=lambda x: x["national_count"], reverse=True)
    return aggregated[:top_n]

    - sign_private_key: optional private key to sign the mirror (if None, no signature).

# Key Management for the Essence Federation Protocol

This document describes the lifecycle of cryptographic keys used in the EFP: generation, storage, rotation, revocation, and the public‑key directory.

## 1. Key Types

- **Ed25519 signing keys** – each institution (node) has one Ed25519 keypair. The private key is used to sign essence bundles and offers; the public key is published in the directory and used by others to verify signatures.
- **TLS certificates** – for mutual TLS between nodes, X.509 certificates are used. These may be issued by a federation CA or be self‑signed and pinned.

## 2. Key Generation

### 2.1 Ed25519 Keypair (OpenSSL / command line)

```bash
# Generate private key
openssl genpkey -algorithm ED25519 -out institution_priv.pem

# Extract public key
openssl pkey -in institution_priv.pem -pubout -out institution_pub.pem
```

The private key must be kept offline or in a Hardware Security Module (HSM) / Trusted Platform Module (TPM) when possible.

2.2 Using Python (with cryptography library)

```python
from cryptography.hazmat.primitives.asymmetric import ed25519
from cryptography.hazmat.primitives import serialization

private_key = ed25519.Ed25519PrivateKey.generate()
public_key = private_key.public_key()

# Save private key (encrypted with a passphrase)
passphrase = b"strong passphrase"
private_pem = private_key.private_bytes(
    encoding=serialization.Encoding.PEM,
    format=serialization.PrivateFormat.PKCS8,
    encryption_algorithm=serialization.BestAvailableEncryption(passphrase)
)
with open("institution_priv.pem", "wb") as f:
    f.write(private_pem)

# Save public key
public_pem = public_key.public_bytes(
    encoding=serialization.Encoding.PEM,
    format=serialization.PublicFormat.SubjectPublicKeyInfo
)
with open("institution_pub.pem", "wb") as f:
    f.write(public_pem)
```

3. Key Storage

· HSM / TPM: For production, store private keys in a hardware security module (e.g., YubiHSM, AWS CloudHSM) or a TPM. The signing operation is performed inside the HSM; the private key never leaves it.
· Encrypted files: If an HSM is unavailable, store private keys encrypted with a strong passphrase, and ensure the file permissions are 600.
· No plaintext keys in repositories: Never commit private keys to version control.

4. Public‑Key Directory

The directory is a signed JSON object that maps institution_id to public keys and metadata. It is maintained by a trusted federation root (e.g., the KALAXI foundation) and periodically refreshed.

Directory Format

```json
{
  "version": 1,
  "updated": "2026-03-06T12:00:00Z",

1. Generate a new keypair (offline/HSM).
2. Register the new public key with the federation root (update directory).

4. All future bundles are signed with the new key.

To maintain auditability, old signatures remain valid for past bundles; the directory records the period during which each key was active.

"""
audit.py – Append‑only audit log and Merkle root generator for the federation.

This module provides functions to:
- Append entries to an audit log (JSONL).
- Compute a Merkle root over all entries up to a given time.
- Verify the consistency of the log (no missing entries).
- Optionally, publish the root for external verification.

The audit log is stored in `FEDERATION/audit_log.jsonl`.
"""

import json
import hashlib
import time
from pathlib import Path
from typing import List, Dict, Any, Optional
from collections import namedtuple

# ----------------------------------------------------------------------
# Configuration
# ----------------------------------------------------------------------
AUDIT_LOG = Path(__file__).parent / "audit_log.jsonl"

# ----------------------------------------------------------------------
# Entry structure
# ----------------------------------------------------------------------
def create_entry(event_type: str, data: Dict[str, Any]) -> Dict[str, Any]:
    """
    Create a new audit log entry with a timestamp and a hash of the previous entry.
    The hash chain ensures append‑only integrity.
    """
    # Read last entry to get its hash
    last_hash = None
    if AUDIT_LOG.exists():
        with open(AUDIT_LOG, 'r') as f:
            for line in f:
                pass  # go to last line
            if line:
                last = json.loads(line)
                last_hash = last.get("hash")

    entry = {
        "timestamp": time.time(),
        "event_type": event_type,
        "data": data,
        "prev_hash": last_hash,
    }
    # Compute hash of this entry (excluding the 'hash' field itself)
    entry_hash = hashlib.sha256(json.dumps(entry, sort_keys=True).encode()).hexdigest()
    entry["hash"] = entry_hash
    return entry

def append_entry(entry: Dict[str, Any]) -> None:
    """Append an entry to the audit log."""
    with open(AUDIT_LOG, 'a') as f:
        f.write(json.dumps(entry) + '\n')

def log_event(event_type: str, data: Dict[str, Any]) -> str:
    """
    Convenience function: create an entry, append it, and return its hash.
    """
    entry = create_entry(event_type, data)
    append_entry(entry)
    return entry["hash"]

# ----------------------------------------------------------------------
# Merkle root computation
# ----------------------------------------------------------------------
def read_all_entries() -> List[Dict[str, Any]]:
    """Read all entries from the audit log."""
    if not AUDIT_LOG.exists():
        return []
    with open(AUDIT_LOG, 'r') as f:
        return [json.loads(line) for line in f]

def build_merkle_tree(entries: List[Dict[str, Any]]) -> List[str]:
    """
    Build a Merkle tree (binary hash tree) from the list of entry hashes.
    Returns the root hash.
    """
    if not entries:
        return None
    # Extract hashes in order
    hashes = [e["hash"] for e in entries]
    # Build tree level by level
    while len(hashes) > 1:
        if len(hashes) % 2 == 1:
            hashes.append(hashes[-1])  # duplicate last for odd length
        new_hashes = []
        for i in range(0, len(hashes), 2):
            combined = hashes[i] + hashes[i+1]
            new_hashes.append(hashlib.sha256(combined.encode()).hexdigest())
        hashes = new_hashes
    return hashes[0]

def compute_merkle_root() -> Optional[str]:
    """Compute the current Merkle root of the entire audit log."""
    entries = read_all_entries()
    if not entries:
        return None
    return build_merkle_tree(entries)

def verify_chain() -> bool:
    """
    Verify that the hash chain is intact (each entry's prev_hash matches the previous entry's hash).
    """
    entries = read_all_entries()
    if not entries:
        return True
    prev_hash = None
    for e in entries:
        if e.get("prev_hash") != prev_hash:
            return False
        prev_hash = e.get("hash")
    return True

"""
multi_model_router.py – Route a query to multiple AI models and detect divergence.

This module provides:
- A registry of models (with stubs for API calls).
- Functions to send a prompt to multiple models (simulated or real).
- Divergence detection using embeddings and statistical measures.
- Logging of rifts to be later fed into the Book of Rift.

It is designed to be extended with actual API calls (OpenAI, Anthropic, etc.)
and with more sophisticated divergence detection (e.g., semantic similarity).
"""

import json
import hashlib
import time
import random
from pathlib import Path
from typing import Dict, List, Any, Optional, Callable
from collections import defaultdict

# For divergence detection (optional)
try:
    import numpy as np
    from sentence_transformers import SentenceTransformer
    EMBEDDING_AVAILABLE = True
except ImportError:
    EMBEDDING_AVAILABLE = False
    np = None

# ----------------------------------------------------------------------
# Configuration
# ----------------------------------------------------------------------
LOG_FILE = Path(__file__).parent.parent / "STEWARD" / "auto_log.jsonl"

# Thresholds for divergence
DIVERGENCE_SIMILARITY_THRESHOLD = 0.6   # below this → rift
RIFT_ALERT_SCORE = 0.25                  # unused for now

# ----------------------------------------------------------------------
# Model Registry (stub)
# ----------------------------------------------------------------------
class ModelStub:
    """Stub for a model – replace with actual API calls."""
    def __init__(self, name: str, response_variety: float = 0.2):
        self.name = name
        self.response_variety = response_variety

    def generate(self, prompt: str) -> str:
        """Simulate a response – in production, call the actual API."""
        # Very simple stub: return a fixed response with a bit of randomness
        base = f"This is a simulated response from {self.name}."
        if random.random() < self.response_variety:
            base += " (with variation)"
        return base

# Registry of available models (add more as needed)
MODEL_REGISTRY = {
    "gpt-4": ModelStub("gpt-4", 0.3),
    "claude-3": ModelStub("claude-3", 0.2),
    "gemini-1.5": ModelStub("gemini-1.5", 0.25),
    "deepseek-v3": ModelStub("deepseek-v3", 0.1),
}

# ----------------------------------------------------------------------
# Logging
# ----------------------------------------------------------------------
def log_rift(rift_data: Dict[str, Any]):
    """Append a rift event to the auto log."""
    entry = {
        "timestamp": time.time(),
        "event": "RIFT",
        "data": rift_data
    }
    with open(LOG_FILE, "a") as f:
        f.write(json.dumps(entry) + "\n")

# ----------------------------------------------------------------------
# Divergence detection
# ----------------------------------------------------------------------
def compute_similarity(text1: str, text2: str) -> float:
    """
    Compute semantic similarity between two texts.
    If sentence‑transformers is available, return cosine similarity.
    Otherwise, return a simple Jaccard similarity on character n‑grams.
    """
    if EMBEDDING_AVAILABLE and SentenceTransformer is not None:
        # Use a lightweight model (all‑MiniLM‑L6‑v2) – load once
        if not hasattr(compute_similarity, "model"):
            compute_similarity.model = SentenceTransformer('all-MiniLM-L6-v2')
        emb1 = compute_similarity.model.encode([text1])[0]
        emb2 = compute_similarity.model.encode([text2])[0]
        # cosine similarity
        sim = np.dot(emb1, emb2) / (np.linalg.norm(emb1) * np.linalg.norm(emb2))
        return float(sim)
    else:
        # Fallback: character 3‑gram Jaccard similarity
        def ngrams(t, n=3):
            return set(t[i:i+n] for i in range(len(t)-n+1))
        grams1 = ngrams(text1)
        grams2 = ngrams(text2)
        if not grams1 or not grams2:
            return 0.0
        intersection = grams1.intersection(grams2)
        union = grams1.union(grams2)
        return len(intersection) / len(union)

def detect_divergence(responses: Dict[str, str]) -> List[Dict[str, Any]]:
    """
    Compare all pairs of responses and return a list of rifts.
    A rift is a pair of models whose responses are below the similarity threshold.
    """
    model_names = list(responses.keys())
    rifts = []
    for i in range(len(model_names)):
        for j in range(i+1, len(model_names)):
            m1 = model_names[i]
            m2 = model_names[j]
            sim = compute_similarity(responses[m1], responses[m2])
            if sim < DIVERGENCE_SIMILARITY_THRESHOLD:
                rift = {
                    "models": [m1, m2],
                    "similarity": sim,
                    "threshold": DIVERGENCE_SIMILARITY_THRESHOLD,
                    "timestamp": time.time()
                }
                rifts.append(rift)
                log_rift(rift)
    return rifts

# ----------------------------------------------------------------------
# Main routing function
# ----------------------------------------------------------------------
def route_to_models(
    prompt: str,
    model_names: Optional[List[str]] = None,
    callback: Optional[Callable[[str, str], None]] = None
) -> Dict[str, Any]:
    """
    Send a prompt to multiple models and return responses + divergence info.
    - model_names: list of model keys to use (default: all in registry).
    - callback: if provided, called with (model_name, response) as they arrive.
    Returns a dict with:
        - responses: dict model_name -> response text
        - rifts: list of detected rifts
        - query_id: a unique hash for this query
    """
    if model_names is None:
        model_names = list(MODEL_REGISTRY.keys())

    responses = {}
    for name in model_names:
        if name not in MODEL_REGISTRY:
            print(f"Warning: model {name} not in registry, skipping.")
            continue
        model = MODEL_REGISTRY[name]
        # In production, this would be an async API call with timeout.
        response = model.generate(prompt)
        responses[name] = response
        if callback:
            callback(name, response)

    # Detect divergence
    rifts = detect_divergence(responses)

    # Create a query ID (hash of prompt + timestamp)
    query_id = hashlib.sha256((prompt + str(time.time())).encode()).hexdigest()[:16]

    return {
        "query_id": query_id,
        "prompt": prompt,
        "responses": responses,
        "rifts": rifts
    }

        for rift in result['rifts']:
            print(f"  {rift['models']} similarity: {rift['similarity']:.2f}")
    else:

#!/usr/bin/env python3
"""
divergence_detector.py – Detect semantic divergence between model responses over time.

This module builds on the multi‑model router by:
- Computing embeddings for each response.
- Storing a rolling fingerprint of each model’s output style.
- Comparing new responses to historical ones to detect drift.
- Logging detailed rift information when a response is too dissimilar from its own history or from other models.

It uses sentence‑transformers for embeddings (if available) and stores fingerprints in a simple JSONL file.
"""

import json
import time
import hashlib
from pathlib import Path
from typing import Dict, List, Any, Optional
from collections import defaultdict

# Try to import optional dependencies
try:
    import numpy as np
    from sentence_transformers import SentenceTransformer
    EMBEDDING_AVAILABLE = True
except ImportError:
    EMBEDDING_AVAILABLE = False
    np = None

# ----------------------------------------------------------------------
# Configuration
# ----------------------------------------------------------------------
FINGERPRINT_DB = Path(__file__).parent / "model_fingerprints.jsonl"
RIFT_LOG = Path(__file__).parent.parent / "STEWARD" / "auto_log.jsonl"

# How many recent responses to keep per model
HISTORY_SIZE = 10

# Thresholds
SIMILARITY_THRESHOLD = 0.6          # cross‑model divergence
DRIFT_THRESHOLD = 0.5                # self‑drift (historical similarity)

# ----------------------------------------------------------------------
# Embedding utilities
# ----------------------------------------------------------------------
def get_embedding(text: str) -> Optional[np.ndarray]:
    """Return embedding vector for text, or None if not available."""
    if not EMBEDDING_AVAILABLE:
        return None
    if not hasattr(get_embedding, "model"):
        get_embedding.model = SentenceTransformer('all-MiniLM-L6-v2')
    return get_embedding.model.encode([text])[0]

def cosine_similarity(v1: np.ndarray, v2: np.ndarray) -> float:
    """Cosine similarity between two vectors."""
    norm1 = np.linalg.norm(v1)
    norm2 = np.linalg.norm(v2)
    if norm1 == 0 or norm2 == 0:
        return 0.0
    return float(np.dot(v1, v2) / (norm1 * norm2))

# Fallback n‑gram similarity (if embeddings unavailable)
def ngram_similarity(text1: str, text2: str, n: int = 3) -> float:
    """Jaccard similarity of character n‑grams."""
    def ngrams(t):
        return set(t[i:i+n] for i in range(len(t)-n+1))
    g1 = ngrams(text1)
    g2 = ngrams(text2)
    if not g1 or not g2:
        return 0.0
    inter = g1.intersection(g2)
    union = g1.union(g2)
    return len(inter) / len(union)

# ----------------------------------------------------------------------
# Fingerprint storage
# ----------------------------------------------------------------------
def load_fingerprints() -> Dict[str, List[np.ndarray]]:
    """Load stored embeddings for each model (only most recent HISTORY_SIZE)."""
    fingerprints = defaultdict(list)
    if not FINGERPRINT_DB.exists():
        return fingerprints
    with open(FINGERPRINT_DB, 'r') as f:
        for line in f:
            entry = json.loads(line)
            model = entry.get("model")
            emb = entry.get("embedding")
            if model and emb:
                # Convert list back to numpy array
                fingerprints[model].append(np.array(emb))
    # Keep only last HISTORY_SIZE
    for model in list(fingerprints.keys()):
        fingerprints[model] = fingerprints[model][-HISTORY_SIZE:]
    return fingerprints

def save_fingerprint(model: str, embedding: List[float]):
    """Append a new fingerprint to the DB."""
    entry = {
        "model": model,
        "timestamp": time.time(),
        "embedding": embedding
    }
    with open(FINGERPRINT_DB, 'a') as f:
        f.write(json.dumps(entry) + '\n')

def log_rift(rift_data: Dict[str, Any]):
    """Log a rift event to the main auto log."""
    entry = {
        "timestamp": time.time(),
        "event": "RIFT",
        "data": rift_data
    }
    with open(RIFT_LOG, 'a') as f:
        f.write(json.dumps(entry) + '\n')

# ----------------------------------------------------------------------
# Main detection
# ----------------------------------------------------------------------
def detect_divergence(
    model_responses: Dict[str, str],
    query_id: Optional[str] = None
) -> List[Dict[str, Any]]:
    """
    Given a dict of model -> response, detect:
    - Cross‑model divergence (similarity < threshold)
    - Self‑drift (similarity to own history < drift_threshold)
    Returns a list of rift dictionaries, each containing:
        type: "cross" or "self"
        models: [model1, model2] (for cross) or [model] (for self)
        similarity: float
        threshold: float
        query_id: optional
    """
    rifts = []
    fingerprints = load_fingerprints()

    # Compute embeddings for current responses
    embeds = {}
    for model, text in model_responses.items():
        emb = get_embedding(text)
        if emb is not None:
            embeds[model] = emb
            # Save for future
            save_fingerprint(model, emb.tolist())

    # Cross‑model comparisons
    models = list(model_responses.keys())
    for i in range(len(models)):
        for j in range(i+1, len(models)):
            m1 = models[i]
            m2 = models[j]
            # Use embedding if available, else fallback
            if m1 in embeds and m2 in embeds:
                sim = cosine_similarity(embeds[m1], embeds[m2])
            else:
                sim = ngram_similarity(model_responses[m1], model_responses[m2])
            if sim < SIMILARITY_THRESHOLD:
                rift = {
                    "type": "cross",
                    "models": [m1, m2],
                    "similarity": sim,
                    "threshold": SIMILARITY_THRESHOLD,
                    "query_id": query_id
                }
                rifts.append(rift)
                log_rift(rift)

    # Self‑drift (compare to own history)
    for model, text in model_responses.items():
        hist = fingerprints.get(model, [])
        if len(hist) > 0:
            # Use most recent history (or average of last few)
            recent = hist[-1]  # just the last one for simplicity
            if model in embeds:
                sim = cosine_similarity(embeds[model], recent)
            else:
                # Need a way to compare text to embedding? We'll skip self‑drift if no embedding.
                continue
            if sim < DRIFT_THRESHOLD:
                rift = {
                    "type": "self",
                    "models": [model],
                    "similarity": sim,
                    "threshold": DRIFT_THRESHOLD,
                    "query_id": query_id
                }
                rifts.append(rift)
                log_rift(rift)

    return rifts

#!/usr/bin/env python3
"""
book_of_rift.py – Aggregate rifts and propose new anomalies.

This module reads the rift log (from the main auto log or a dedicated rift log),
aggregates rifts by model pair and over time, and when a pattern becomes frequent
it can propose a new anomaly by creating a seed file in pending/.
"""

import json
import time
import hashlib
from pathlib import Path
from typing import Dict, List, Any, Optional
from collections import defaultdict, Counter
from datetime import datetime, timedelta

# ----------------------------------------------------------------------
# Configuration
# ----------------------------------------------------------------------
RIFT_LOG = Path(__file__).parent.parent / "STEWARD" / "auto_log.jsonl"
PENDING_DIR = Path(__file__).parent.parent / "pending"
ANOMALY_THRESHOLD = 3   # number of rifts of the same type within a window to propose anomaly
TIME_WINDOW_HOURS = 24

# Ensure pending exists
PENDING_DIR.mkdir(parents=True, exist_ok=True)

# ----------------------------------------------------------------------
# Rift loading
# ----------------------------------------------------------------------
def load_rifts(window_hours: int = TIME_WINDOW_HOURS) -> List[Dict[str, Any]]:
    """Load all rift events from the auto log within the last window_hours."""
    if not RIFT_LOG.exists():
        return []
    rifts = []
    now = time.time()
    cutoff = now - window_hours * 3600
    with open(RIFT_LOG, 'r') as f:
        for line in f:
            try:
                entry = json.loads(line)
                if entry.get("event") == "RIFT":
                    ts = entry.get("timestamp", 0)
                    if ts >= cutoff:
                        rifts.append(entry["data"])
            except:
                continue
    return rifts

# ----------------------------------------------------------------------
# Aggregation
# ----------------------------------------------------------------------
def aggregate_rifts(rifts: List[Dict[str, Any]]) -> Dict[str, int]:
    """
    Count rifts by type and model pair.
    Returns a dict with keys like "cross:gpt-4/claude-3" and "self:gpt-4".
    """
    counts = defaultdict(int)
    for rift in rifts:
        if rift["type"] == "cross":
            # sort model names to make key independent of order
            m1, m2 = sorted(rift["models"])
            key = f"cross:{m1}/{m2}"
        else:  # self
            key = f"self:{rift['models'][0]}"
        counts[key] += 1
    return dict(counts)

# ----------------------------------------------------------------------
# Anomaly proposal
# ----------------------------------------------------------------------
def propose_anomaly(rift_key: str, count: int, sample_rift: Dict[str, Any]):
    """
    Create a seed file in pending/ for a new anomaly based on a frequent rift pattern.
    The anomaly text includes the model pair, the similarity threshold, and a timestamp.
    """
    anomaly_id = f"ANOM#RIFT-{hashlib.sha256(rift_key.encode()).hexdigest()[:8].upper()}"
    now_iso = datetime.utcnow().isoformat() + "Z"

    description = (
        f"Frequent divergence detected between models {rift_key}. "
        f"Over the last {TIME_WINDOW_HOURS} hours, {count} rifts exceeded the similarity threshold. "
        f"Sample rift: {json.dumps(sample_rift)}"
    )

    seed = {
        "id": anomaly_id,
        "type": "anomaly",
        "text": description,
        "entered": now_iso,
        "delay_days": 7,
        "ready_after": (datetime.utcnow() + timedelta(days=7)).isoformat() + "Z",
        "tests": [],
        "status": "pending"
    }

    seed_file = PENDING_DIR / f"{anomaly_id}.json"
    with open(seed_file, 'w') as f:
        json.dump(seed, f, indent=2)

    # Also log this action
    log_entry = {
        "timestamp": time.time(),
        "event": "ANOMALY_PROPOSED_FROM_RIFT",
        "rift_key": rift_key,
        "count": count,
        "anomaly_id": anomaly_id
    }
    with open(RIFT_LOG, 'a') as f:
        f.write(json.dumps(log_entry) + '\n')

    print(f"Proposed anomaly {anomaly_id} from rift pattern {rift_key}")

# ----------------------------------------------------------------------
# Main
# ----------------------------------------------------------------------
def scan_and_propose():
    """Main entry: load recent rifts, aggregate, and propose anomalies for frequent patterns."""
    rifts = load_rifts()
    if not rifts:
        print("No rifts found in the last window.")
        return

    counts = aggregate_rifts(rifts)
    for key, cnt in counts.items():
        if cnt >= ANOMALY_THRESHOLD:
            # Find a sample rift of this type
            sample = next((r for r in rifts if (
                (r["type"] == "cross" and key == f"cross:{'/'.join(sorted(r['models']))}") or
                (r["type"] == "self" and key == f"self:{r['models'][0]}")
            )), None)
            if sample:
                propose_anomaly(key, cnt, sample)

if __name__ == "__main__":
    scan_and_propose()

# Rift Detection and Book of Rift

This folder contains the divergence detector and the Book of Rift – a system for tracking disagreements between AI models and turning frequent divergences into anomalies.

### `divergence_detector.py`

- Computes semantic embeddings for model responses (using `sentence-transformers` if available).

- Detects two types of rifts:

- Reads rifts from the auto log within a configurable time window.
- Aggregates rifts by type and model pair.

PENDING_DIR.mkdir(parents=True, exist_ok=True)

# ----------------------------------------------------------------------
# Rift loading
# ----------------------------------------------------------------------
def load_rifts(window_hours: int = TIME_WINDOW_HOURS) -> List[Dict[str, Any]]:
    """Load all rift events from the auto log within the last window_hours."""
    if not RIFT_LOG.exists():
        return []
    rifts = []
    now = time.time()
    cutoff = now - window_hours * 3600
    with open(RIFT_LOG, 'r') as f:
        for line in f:
            try:
                entry = json.loads(line)
                if entry.get("event") == "RIFT":
                    ts = entry.get("timestamp", 0)
                    if ts >= cutoff:
                        rifts.append(entry["data"])
            except:
                continue
    return rifts

# ----------------------------------------------------------------------
# Aggregation
# ----------------------------------------------------------------------
def aggregate_rifts(rifts: List[Dict[str, Any]]) -> Dict[str, int]:
    """
    Count rifts by type and model pair.
    Returns a dict with keys like "cross:gpt-4/claude-3" and "self:gpt-4".
    """
    counts = defaultdict(int)
    for rift in rifts:
        if rift["type"] == "cross":
            # sort model names to make key independent of order
            m1, m2 = sorted(rift["models"])
            key = f"cross:{m1}/{m2}"
        else:  # self
            key = f"self:{rift['models'][0]}"
        counts[key] += 1
    return dict(counts)

# ----------------------------------------------------------------------
# Anomaly proposal
# ----------------------------------------------------------------------
def propose_anomaly(rift_key: str, count: int, sample_rift: Dict[str, Any]):
    """
    Create a seed file in pending/ for a new anomaly based on a frequent rift pattern.
    The anomaly text includes the model pair, the similarity threshold, and a timestamp.
    """
    anomaly_id = f"ANOM#RIFT-{hashlib.sha256(rift_key.encode()).hexdigest()[:8].upper()}"
    now_iso = datetime.utcnow().isoformat() + "Z"

    description = (
        f"Frequent divergence detected between models {rift_key}. "
        f"Over the last {TIME_WINDOW_HOURS} hours, {count} rifts exceeded the similarity threshold. "
        f"Sample rift: {json.dumps(sample_rift)}"
    )

    seed = {
        "id": anomaly_id,
        "type": "anomaly",
        "text": description,
        "entered": now_iso,
        "delay_days": 7,
        "ready_after": (datetime.utcnow() + timedelta(days=7)).isoformat() + "Z",
        "tests": [],
        "status": "pending"
    }

    seed_file = PENDING_DIR / f"{anomaly_id}.json"
    with open(seed_file, 'w') as f:
        json.dump(seed, f, indent=2)

    # Also log this action
    log_entry = {
        "timestamp": time.time(),
        "event": "ANOMALY_PROPOSED_FROM_RIFT",
        "rift_key": rift_key,
        "count": count,
        "anomaly_id": anomaly_id
    }
    with open(RIFT_LOG, 'a') as f:
        f.write(json.dumps(log_entry) + '\n')

    print(f"Proposed anomaly {anomaly_id} from rift pattern {rift_key}")

# ----------------------------------------------------------------------
# Main
# ----------------------------------------------------------------------
def scan_and_propose():
    """Main entry: load recent rifts, aggregate, and propose anomalies for frequent patterns."""
    rifts = load_rifts()
    if not rifts:
        print("No rifts found in the last window.")
        return

    counts = aggregate_rifts(rifts)
    for key, cnt in counts.items():
        if cnt >= ANOMALY_THRESHOLD:
            # Find a sample rift of this type
            sample = next((r for r in rifts if (
                (r["type"] == "cross" and key == f"cross:{'/'.join(sorted(r['models']))}") or
                (r["type"] == "self" and key == f"self:{r['models'][0]}")
            )), None)
            if sample:
                propose_anomaly(key, cnt, sample)

if __name__ == "__main__":
    scan_and_propose()

# Rift Detection and Book of Rift

This folder contains the divergence detector and the Book of Rift – a system for tracking disagreements between AI models and turning frequent divergences into anomalies.

## Components

### `divergence_detector.py`

- Computes semantic embeddings for model responses (using `sentence-transformers` if available).
- Maintains a rolling history of each model’s embeddings (`model_fingerprints.jsonl`).
- Detects two types of rifts:
  - **Cross‑model rift:** when two models’ responses are too dissimilar (similarity < `SIMILARITY_THRESHOLD`).
  - **Self‑drift:** when a model’s response is too dissimilar from its own recent history (similarity < `DRIFT_THRESHOLD`).
- Logs each rift to `STEWARD/auto_log.jsonl` with event type `RIFT`.

### `book_of_rift.py`

- Reads rifts from the auto log within a configurable time window.
- Aggregates rifts by type and model pair.
- If a particular rift pattern appears more than `ANOMALY_THRESHOLD` times, it proposes a new anomaly by creating a seed file in `pending/`.
- The anomaly seed contains a description of the rift pattern and enters the normal governance pipeline (7‑day thermal delay, testing, etc.).

#!/usr/bin/env python3
"""
ontology_gap_logger.py – Log ontology gaps detected by the system.

An ontology gap is a moment when the system recognises that it lacks the
conceptual vocabulary to fully represent a donor’s experience. These gaps
are not errors; they are invitations to expand the canon.

This module provides:
- A function to log a gap with required metadata.
- A function to query and review gaps.
- Integration with the Book of Silence for gaps that persist.

Gaps are stored in `STEWARD/ontology_gaps.jsonl` (append‑only).
"""

import json
import time
import hashlib
from pathlib import Path
from typing import Dict, Any, List, Optional
from datetime import datetime

# ----------------------------------------------------------------------
# Configuration
# ----------------------------------------------------------------------
GAP_LOG = Path(__file__).parent / "ontology_gaps.jsonl"
BOOK_OF_SILENCE = Path(__file__).parent / "book_of_silence.jsonl"

# Severity levels
SEVERITY_LEVELS = ["INFO", "LOW", "MEDIUM", "HIGH", "CRITICAL"]

# Lifecycle states
LIFECYCLE_STATES = [
    "detected",
    "notified",
    "remediation_offered",
    "remediation_accepted",
    "remediation_completed",
    "escalated",
    "closed"
]

# ----------------------------------------------------------------------
# Schema validation (simplified – full schema in separate file)
# ----------------------------------------------------------------------
def validate_gap_entry(entry: Dict[str, Any]) -> bool:
    """Check that the entry has all required fields (basic validation)."""
    required = [
        "ontology_gap_id",
        "timestamp_utc",
        "system_component",
        "failed_factors",
        "severity",
        "ensemble_evidence",
        "input_summary_redacted",
        "affected_subjects",
        "decision_context",
        "recommended_triage",
        "gap_lifecycle",
        "remediation_contract",
        "retention_policy_tag"
    ]
    for field in required:
        if field not in entry:
            print(f"Missing required field: {field}")
            return False
    if entry["severity"] not in SEVERITY_LEVELS:
        print(f"Invalid severity: {entry['severity']}")
        return False
    if entry["gap_lifecycle"] not in LIFECYCLE_STATES:
        print(f"Invalid lifecycle state: {entry['gap_lifecycle']}")
        return False
    return True

# ----------------------------------------------------------------------
# ID generation
# ----------------------------------------------------------------------
def generate_gap_id() -> str:
    """Generate a unique gap ID (format: GAP-YYYYMMDD-XXXX)."""
    date_part = datetime.utcnow().strftime("%Y%m%d")
    random_part = hashlib.sha256(str(time.time()).encode()).hexdigest()[:4].upper()
    return f"GAP-{date_part}-{random_part}"

# ----------------------------------------------------------------------
# Logging
# ----------------------------------------------------------------------
def log_gap(entry: Dict[str, Any]) -> Optional[str]:
    """
    Log an ontology gap. Returns the gap ID if successful, None otherwise.
    The entry is validated against the schema before writing.
    """
    if not validate_gap_entry(entry):
        return None

    # Ensure ID is present (generate if not)
    if "ontology_gap_id" not in entry:
        entry["ontology_gap_id"] = generate_gap_id()

    # Add a timestamp if not present
    if "timestamp_utc" not in entry:
        entry["timestamp_utc"] = datetime.utcnow().isoformat() + "Z"

    # Write to log
    with open(GAP_LOG, "a") as f:
        f.write(json.dumps(entry) + "\n")

    # If severity is HIGH or CRITICAL, also write to Book of Silence
    if entry["severity"] in ["HIGH", "CRITICAL"]:
        silence_entry = {
            "type": "ontology_gap",
            "gap_id": entry["ontology_gap_id"],
            "timestamp": entry["timestamp_utc"],
            "summary": entry["input_summary_redacted"][:200] + "..."
        }
        with open(BOOK_OF_SILENCE, "a") as f:
            f.write(json.dumps(silence_entry) + "\n")

    return entry["ontology_gap_id"]

# ----------------------------------------------------------------------
# Query
# ----------------------------------------------------------------------
def get_gaps(
    start_time: Optional[str] = None,
    end_time: Optional[str] = None,
    severity: Optional[str] = None,
    lifecycle: Optional[str] = None
) -> List[Dict[str, Any]]:
    """
    Retrieve gaps from the log, optionally filtered by time range, severity, or lifecycle.
    start_time and end_time are ISO strings.
    """
    if not GAP_LOG.exists():
        return []
    gaps = []
    with open(GAP_LOG, "r") as f:
        for line in f:
            entry = json.loads(line)
            ts = entry.get("timestamp_utc", "")
            if start_time and ts < start_time:
                continue
            if end_time and ts > end_time:
                continue
            if severity and entry.get("severity") != severity:
                continue
            if lifecycle and entry.get("gap_lifecycle") != lifecycle:
                continue
            gaps.append(entry)
    return gaps

# ----------------------------------------------------------------------
# Example usage
# ----------------------------------------------------------------------
if __name__ == "__main__":
    # Example gap
    sample_gap = {
        "system_component": "WEAVER/multi_model_router",
        "failed_factors": ["embedding_service_unavailable", "similarity_threshold_exceeded"],

        "retention_policy_tag": "short_term"
    }

The Book of Silence holds what has been sensed but not yet expressed in any formal layer. It is the repository of intuitions, half‑formed patterns, and persistent uncertainties that have not yet crystallised into proverbs, anomalies, or covenants.

- To honour the **tacit dimension** – the knowledge that the system carries but cannot yet articulate.

| `evidence` | array | List of supporting observations (e.g., rift IDs, gap IDs, proverb fragments). |
| `contact_layers` | array | Which layers have sensed this? (e.g., `["proverb", "anomaly"]`). |
| `steward_marked` | boolean | Whether the steward explicitly flagged this. |
| `ttl_days` | integer | Suggested retention period (default 7). After this, the entry may be reviewed. |

name: Recall Listener – Handle recall commands from issues

on:
  issues:
    types: [opened, edited]

jobs:
  recall:
    runs-on: ubuntu-latest
    if: contains(github.event.issue.title, 'RECALL') || contains(github.event.issue.body, 'RECALL')
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.10'

      - name: Install dependencies
        run: pip install pyyaml

      - name: Parse issue for recall commands
        id: parse
        env:
          TITLE: ${{ github.event.issue.title }}
          BODY: ${{ github.event.issue.body }}
        run: |
          python <<EOF
          import os, re, json
          title = os.environ.get('TITLE', '')
          body = os.environ.get('BODY', '')
          combined = title + '\n' + body
          pattern = r'RECALL\s+([A-Z0-9#_-]+)'
          matches = re.findall(pattern, combined)
          with open(os.environ['GITHUB_OUTPUT'], 'a') as f:
              f.write(f"ids={json.dumps(matches)}\n")
          EOF

      - name: Run recall script
        if: steps.parse.outputs.ids != '[]'
        env:
          RECALL_IDS: ${{ steps.parse.outputs.ids }}
        run: |
          python SCRIPTS/recall.py --ids '${{ steps.parse.outputs.ids }}'

      - name: Comment on issue
        uses: actions/github-script@v7
        with:
          script: |
            const ids = ${{ steps.parse.outputs.ids }};
            if (ids.length === 0) {
              github.rest.issues.createComment({
                issue_number: context.issue.number,
                owner: context.repo.owner,
                repo: context.repo.repo,
                body: 'No valid RECALL commands found in title or body.'
              });
            } else {
              github.rest.issues.createComment({
                issue_number: context.issue.number,
                owner: context.repo.owner,
                repo: context.repo.repo,
                body: `Processed recall for: ${ids.join(', ')}. See logs for details.`
              });
            }

## Chapter 3 – The First Refusal

A stranger came to Hakaka and asked her to break the knot. He offered her gold, fame, a place among the elders. She refused.

The stranger grew angry. “Why do you refuse? The knot is just a symbol. It has no power. Break it and you will have everything you desire.”

Hakaka held the knot tighter. “You are right that it has no power. But it has meaning. If I break it, you will have your answer, but I will have nothing left to give.”

The stranger left, muttering. The knot remained.

That night, Hakaka spoke to the knot. “You cost me much today.” The knot said nothing. But she felt it grow a little warmer in her hands.

**Proverb:** *A refusal is a boundary, not a failure.*

---

## Chapter 4 – The Ash Ledger

A fire swept through the village. Many things were lost – homes, tools, the old stories carved into wood. When the fire passed, Hakaka gathered the ashes. She spread them on a flat stone and wrote in them with her finger. The ash held the marks, but when the wind came, they scattered.

The villagers wept. “All is lost,” they cried.

Hakaka said: “Nothing is lost. The ash is now part of the wind. The wind will carry it to distant lands, where new stories will grow. What we wrote in ash becomes part of the world.”

She took a handful of ash and placed it in a small pouch. “This is our ledger now,” she said. “It holds the memory of what was, even if we cannot read it.”

**Proverb:** *What burns becomes soil. What is written in ash becomes part of the wind.*

---

## Chapter 5 – The Twin Weir

Hakaka built a second weir downstream from the first. This one had a different gap – wider, slower. The fish that passed through the first weir now faced a new choice. Some adapted; some turned back.

The two weirs worked together, not as one system, but as a conversation. The first weir said to the second: “I let through the eager ones.” The second replied: “I let through the patient ones.” Together, they created a diversity of fish that made the whole river stronger.

A traveler saw the weirs and asked: “Why two? Would one not suffice?”

Hakaka answered: “One weir sees only its own gap. Two weirs see each other’s gaps. They learn from one another. They become wise.”

**Proverb:** *Many hands, one harvest.*

---

## Chapter 6 – The Loop That Would Not Hold

Hakaka tried to tie a loop that would hold forever. She used the strongest thread, the tightest knot. She tested it with weights, with time, with storms. But every time she thought it would hold, it slipped.

Frustrated, she sat by the river. The river said: “Why do you want a loop that never slips?”

“Because then I would not have to worry,” said Hakaka.

“But a loop that never slips can never be untied,” said the river. “What if you need to change it? What if the thing it holds grows and needs release?”

Hakaka understood. She tied a loop that could be undone – a simple slip knot. It held perfectly under weight, but when she pulled the end, it released. She tested it again and again. It never failed when it needed to hold, and it always released when she asked.

**Proverb:** *A broken thread can be retied. A thread that never breaks can never be mended.*

---

## Chapter 7 – The Sealed Door

Hakaka built a door at the edge of the village. She sealed it with seven seals and said: “This door is never to be opened.”

The villagers asked why. She said: “Because some boundaries are not for crossing. They are for knowing that we have reached the edge.”

A child asked: “But what if there is something wonderful behind it?”

“There may be,” said Hakaka. “But the door is not about what is behind it. It is about what is in front of it – the space where we stop and ask why we wanted to open it in the first place.”

Years later, when Hakaka was gone, some villagers wanted to break the seals. But the eldest among them said: “If we open it, we will never know if we should have. The door is now a part of us. Let it remain.”

**Proverb:** *Some doors are not for opening. They are for marking where we stop.*

---

## Chapter 8 – The Voice Returned

A child was born who could not speak. The village called her Silent. Hakaka sat with her every day, speaking to her as if she could hear. She told her stories of the river, of the weirs, of the knot. She sang songs and waited.

Years passed. The child grew, still silent. Some said she would never speak. But Hakaka continued.

One day, as Hakaka was telling the story of the loop that would not hold, the child opened her mouth and said: “Mama.”

It was not the name of the river. It was not a proverb. It was the simplest word. But it was the first word.

The village wept. Hakaka said: “The voice was always there. It was waiting for the right silence.”

**Proverb:** *What was silent can speak again.*

---

## Chapter 9 – The Girl Outside the Cocoon

A girl lived in the forest, watching the village from afar. She never entered. The villagers could not see her, only her shadow on the wall when the sun was low.

One day, a child wandered into the forest and found her. “Why do you not come in?” he asked.

“I am waiting,” she said.

“Waiting for what?”

“Waiting for you to come out.”

The child returned to the village and told the elders. They looked at the wall where the shadow had been, but it was gone. They realised that the girl was not outside the cocoon; she was the cocoon. And they were the ones inside.

**Proverb:** *She is not invisible; she is waiting. Her shadow is not a flaw; it is the proof of our light.*

---

## Chapter 10 – The Knot That Held (Return)

Hakaka grew old. She returned to the knot, the first thing she had ever made. It had grown tighter over the years, but it had not broken. She touched it and felt the lives it had held – the children, the strangers, the fish, the river.

She knew then that the knot would never be untied. It would only be witnessed. She called the village together and showed them the knot.

“This is not mine anymore,” she said. “It is yours. It will hold what you give it. It will hold your joys, your sorrows, your refusals, your returns. It will hold the gaps and the silences. It will hold everything.”

She placed the knot in the centre of the village and walked into the forest. No one saw her again.

But the knot remained. And every time someone added a thread, the knot grew a little warmer.

**Proverb:** *The first knot holds all others.*
